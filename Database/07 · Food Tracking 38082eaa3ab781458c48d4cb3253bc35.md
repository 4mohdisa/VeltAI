# 07 · Food Tracking

Meals, food items, photo storage, voice transcripts, AI scan logs, corrections, and per-user food memory.

---

## `meals`

One row per meal event.

```sql
CREATE TABLE meals (
  id                  UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  user_id             UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  meal_date           DATE NOT NULL,
  meal_slot           TEXT NOT NULL,        -- breakfast | lunch | dinner | snack | pre_workout | post_workout

  -- Cached totals (computed from food_items)
  total_calories      NUMERIC(7,1) DEFAULT 0,
  total_protein_g     NUMERIC(6,1) DEFAULT 0,
  total_carbs_g       NUMERIC(6,1) DEFAULT 0,
  total_fat_g         NUMERIC(6,1) DEFAULT 0,
  total_fibre_g       NUMERIC(6,1) DEFAULT 0,

  -- Entry source
  entry_method        TEXT NOT NULL,        -- manual | photo_scan | voice | barcode
  notes               TEXT,

  logged_at           TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  created_at          TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at          TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE INDEX idx_meals_user_date ON meals(user_id, meal_date DESC);

ALTER TABLE meals ENABLE ROW LEVEL SECURITY;
CREATE POLICY "meals_own" ON meals
  USING (user_id = (SELECT id FROM users WHERE clerk_id = auth.uid()::TEXT));
```

## `food_items`

One meal can have many food items. e.g. "grilled chicken", "rice", "broccoli".

```sql
CREATE TABLE food_items (
  id                  UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  meal_id             UUID NOT NULL REFERENCES meals(id) ON DELETE CASCADE,
  user_id             UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,

  name                TEXT NOT NULL,
  quantity_g          NUMERIC(7,1),
  quantity_ml         NUMERIC(7,1),
  quantity_units      TEXT,                 -- for items measured as "pieces", "cups", etc.

  calories            NUMERIC(7,1),
  protein_g           NUMERIC(6,1),
  carbs_g             NUMERIC(6,1),
  fat_g               NUMERIC(6,1),
  fibre_g             NUMERIC(6,1),
  sodium_mg           NUMERIC(7,1),

  data_source         TEXT NOT NULL DEFAULT 'manual',
    -- manual | photo_scan | voice | barcode | user_memory | food_db
  off_product_id      TEXT,                 -- Open Food Facts barcode ID
  confidence          NUMERIC(3,2),         -- AI confidence 0.00–1.00 (NULL if manual)
  is_corrected        BOOLEAN DEFAULT FALSE, -- true if user edited the AI's guess

  created_at          TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE INDEX idx_food_items_meal ON food_items(meal_id);
CREATE INDEX idx_food_items_user ON food_items(user_id);

ALTER TABLE food_items ENABLE ROW LEVEL SECURITY;
CREATE POLICY "food_items_own" ON food_items
  USING (user_id = (SELECT id FROM users WHERE clerk_id = auth.uid()::TEXT));
```

## `meal_photos`

Multiple photos per meal: food, nutrition label, leftovers. All linked to the meal.

```sql
CREATE TABLE meal_photos (
  id                  UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  meal_id             UUID NOT NULL REFERENCES meals(id) ON DELETE CASCADE,
  user_id             UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,

  photo_type          TEXT NOT NULL,         -- food | label | leftover | other
  storage_path        TEXT NOT NULL,         -- Supabase Storage path
  thumbnail_path      TEXT,                  -- 200px thumbnail path
  width_px            INT,
  height_px           INT,
  file_size_bytes     BIGINT,
  taken_at            TIMESTAMPTZ,           -- EXIF timestamp if available
  uploaded_at         TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE INDEX idx_meal_photos_meal ON meal_photos(meal_id);

ALTER TABLE meal_photos ENABLE ROW LEVEL SECURITY;
CREATE POLICY "meal_photos_own" ON meal_photos
  USING (user_id = (SELECT id FROM users WHERE clerk_id = auth.uid()::TEXT));
```

## `voice_meal_logs`

Voice-described meals. Stored for replay/audit and to feed the training pipeline.

```sql
CREATE TABLE voice_meal_logs (
  id                  UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  user_id             UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  meal_id             UUID REFERENCES meals(id) ON DELETE SET NULL,

  audio_storage_path  TEXT,                  -- Supabase Storage path to audio file
  audio_duration_sec  INT,
  transcript          TEXT NOT NULL,         -- what user said
  transcript_language TEXT DEFAULT 'en',
  transcript_source   TEXT NOT NULL DEFAULT 'ios_speech',
    -- ios_speech | whisper_api

  gemini_raw_response JSONB,                 -- Gemini's parsing of transcript into food items
  parsing_confidence  NUMERIC(3,2),

  user_confirmed      BOOLEAN DEFAULT FALSE, -- did user accept the parsed items?
  user_edits          JSONB,                 -- what they changed

  created_at          TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE INDEX idx_voice_logs_user ON voice_meal_logs(user_id);
CREATE INDEX idx_voice_logs_meal ON voice_meal_logs(meal_id);

ALTER TABLE voice_meal_logs ENABLE ROW LEVEL SECURITY;
CREATE POLICY "voice_logs_own" ON voice_meal_logs
  USING (user_id = (SELECT id FROM users WHERE clerk_id = auth.uid()::TEXT));
```

## `food_scan_logs`

One row per Gemini photo scan call.

```sql
CREATE TABLE food_scan_logs (
  id                    UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  user_id               UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  meal_id               UUID REFERENCES meals(id) ON DELETE SET NULL,

  scan_type             TEXT NOT NULL,    -- photo_fresh | photo_packaged | barcode
  meal_photo_id         UUID REFERENCES meal_photos(id),

  gemini_raw_response   JSONB,
  uncertainty_items     JSONB,            -- items flagged as low-confidence
  scan_duration_ms      INT,
  model_version         TEXT NOT NULL DEFAULT 'gemini-3.5-flash',

  scanned_at            TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  created_at            TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE INDEX idx_scan_logs_user ON food_scan_logs(user_id);

ALTER TABLE food_scan_logs ENABLE ROW LEVEL SECURITY;
CREATE POLICY "scan_logs_own" ON food_scan_logs
  USING (user_id = (SELECT id FROM users WHERE clerk_id = auth.uid()::TEXT));
```

## `food_scan_corrections`

The training data. Only explicit corrections — never passive approvals.

```sql
CREATE TABLE food_scan_corrections (
  id                    UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  scan_log_id           UUID NOT NULL REFERENCES food_scan_logs(id) ON DELETE CASCADE,
  user_id               UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,

  item_name             TEXT NOT NULL,
  correction_type       TEXT NOT NULL,
    -- oil_type_corrected | oil_quantity_corrected |
    -- ingredient_added | ingredient_removed |
    -- quantity_adjusted | item_name_corrected
  ai_value              JSONB,
  user_value            JSONB,

  corrected_at          TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  anonymised_at         TIMESTAMPTZ,
  included_in_training  BOOLEAN NOT NULL DEFAULT FALSE
);

CREATE INDEX idx_corrections_scan ON food_scan_corrections(scan_log_id);
CREATE INDEX idx_corrections_pending ON food_scan_corrections(included_in_training)
  WHERE included_in_training = FALSE AND anonymised_at IS NOT NULL;

ALTER TABLE food_scan_corrections ENABLE ROW LEVEL SECURITY;
CREATE POLICY "corrections_own" ON food_scan_corrections
  FOR SELECT USING (user_id = (SELECT id FROM users WHERE clerk_id = auth.uid()::TEXT));
```

## `user_food_preferences`

Per-user memory of confirmed ingredient choices. Used to pre-fill uncertainty prompts.

```sql
CREATE TABLE user_food_preferences (
  id                UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  user_id           UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  dish_type         TEXT NOT NULL,        -- "stir fry", "curry", "grilled chicken"
  ingredient_key    TEXT NOT NULL,        -- "cooking_oil", "sauce_type", "protein"
  preferred_value   TEXT NOT NULL,        -- "medium sesame oil", "soy sauce"
  times_confirmed   INT NOT NULL DEFAULT 1,
  last_confirmed_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  created_at        TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at        TIMESTAMPTZ NOT NULL DEFAULT NOW(),

  CONSTRAINT uq_food_pref UNIQUE (user_id, dish_type, ingredient_key)
);

CREATE INDEX idx_food_prefs_user ON user_food_preferences(user_id);

ALTER TABLE user_food_preferences ENABLE ROW LEVEL SECURITY;
CREATE POLICY "food_prefs_own" ON user_food_preferences
  USING (user_id = (SELECT id FROM users WHERE clerk_id = auth.uid()::TEXT));
```

## `daily_nutrition_summary`

Cached daily totals. Updated by Go after each meal log. Used for dashboard and AI plan adherence tracking.

```sql
CREATE TABLE daily_nutrition_summary (
  id                  UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  user_id             UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  summary_date        DATE NOT NULL,
  total_calories      NUMERIC(7,1) NOT NULL DEFAULT 0,
  total_protein_g     NUMERIC(6,1) NOT NULL DEFAULT 0,
  total_carbs_g       NUMERIC(6,1) NOT NULL DEFAULT 0,
  total_fat_g         NUMERIC(6,1) NOT NULL DEFAULT 0,
  total_fibre_g       NUMERIC(6,1) NOT NULL DEFAULT 0,
  calorie_target      INT,                  -- snapshot of target at this date
  protein_target_g    NUMERIC(5,1),
  goal_hit_calories   BOOLEAN,              -- set by midnight cron
  goal_hit_protein    BOOLEAN,
  updated_at          TIMESTAMPTZ NOT NULL DEFAULT NOW(),

  CONSTRAINT uq_daily_nutrition UNIQUE (user_id, summary_date)
);

ALTER TABLE daily_nutrition_summary ENABLE ROW LEVEL SECURITY;
CREATE POLICY "daily_nutrition_own" ON daily_nutrition_summary
  USING (user_id = (SELECT id FROM users WHERE clerk_id = auth.uid()::TEXT));
```
